% Fig. 3 -- why the certificate is NOT a smaller ellipsoid inside a bigger one.
% One inference: C_cert is indefinite, so the estimable object has negative directions
% that Cov(f) can never express.
\begin{tikzpicture}[scale=1.0,every node/.style={font=\small}]
  % ---- (a) C_obs: PSD, spectrum in [0,1] ----
  \begin{scope}
    \node[anchor=south,align=center,font=\small] at (1.95,2.05)
      {\textbf{(a)} $C_{\mathrm{obs}}\succeq 0$};
    \draw[->,>=stealth,gray!70] (0,0) -- (3.95,0) node[below right=1pt,font=\footnotesize] {$j$};
    \draw[->,>=stealth,gray!70] (0,-0.1) -- (0,1.85);
    \node[font=\footnotesize,gray!70,anchor=east] at (-0.05,1.7) {$\lambda_j$};
    \foreach \x/\h in {0.4/1.55,0.9/1.22,1.4/0.90,1.9/0.58,2.4/0.32,2.9/0.14,3.4/0.05}
      \draw[fill=gray!35,draw=gray!60] (\x-0.16,0) rectangle (\x+0.16,\h);
    \node[font=\footnotesize,gray!75,anchor=north] at (1.95,-0.28) {$\lambda_j\in[0,1]$};
  \end{scope}

  % ---- (b) C_cert: indefinite ----
  \begin{scope}[shift={(5.6,0)}]
    \node[anchor=south,align=center,font=\small] at (1.95,2.05)
      {\textbf{(b)} $C_{\mathrm{cert}}(f)$ is \emph{indefinite}};
    \draw[gray!45,dashed] (0,0) -- (3.95,0);
    \draw[->,>=stealth,gray!70] (0,-1.55) -- (3.95,-1.55) node[below right=1pt,font=\footnotesize] {$j$};
    \draw[->,>=stealth,gray!70] (0,-1.65) -- (0,1.85);
    \node[font=\footnotesize,gray!70,anchor=east] at (-0.05,1.7) {$\mu_j$};
    \foreach \x/\h in {0.4/0.62,0.9/0.26,1.4/0.07}
      \draw[fill=accent!30,draw=accent] (\x-0.16,0) rectangle (\x+0.16,\h);
    \foreach \x/\h in {1.9/-0.28,2.4/-0.52,2.9/-0.80,3.4/-1.05}
      \draw[fill=gray!25,draw=gray!60] (\x-0.16,0) rectangle (\x+0.16,\h);
    % labels placed clear of both axes
    \draw[accent,thin] (0.75,0.75) -- (1.35,1.25);
    \node[font=\footnotesize,accent,anchor=west] at (1.30,1.35)
      {$D_{\mathrm{cert}}=\sum_j\max(\mu_j,0)$};
    \node[font=\footnotesize,gray!75,anchor=north] at (2.65,-1.14)
      {decoder misspecification};
  \end{scope}

  \node[font=\footnotesize,align=center,anchor=north] at (4.75,-2.05)
    {$C_{\mathrm{cert}}(f)=C_{\mathrm{obs}}-\Cov(m-f)\preceq C_{\mathrm{obs}}$
     \ orders \emph{quadratic forms}, not volumes.};
\end{tikzpicture}
